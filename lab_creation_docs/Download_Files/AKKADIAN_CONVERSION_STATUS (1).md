### **Step 2: Update lib.rs**

```rust
// akkadian-dsl/src/lib.rs
pub mod ast;
pub mod lexer;
pub mod parser;
pub mod compiler;

pub use ast::*;
pub use lexer::*;
pub use parser::*;

use shared::Result;

/// Parse Akkadian source code into AST
pub fn parse(source: &str) -> Result<AkkadianProgram> {
    let mut parser = parser::AkkadianParser::new(source)
        .map_err(|e| shared::BahyWayError::Parsing(e.to_string()))?;
    parser.parse()
        .map_err(|e| shared::BahyWayError::Parsing(e.to_string()))
}

/// Compile Akkadian program to SQL
pub fn compile_to_sql(program: &AkkadianProgram) -> Result<String> {
    compiler::sql::SqlGenerator::generate(program)
        .map_err(|e| shared::BahyWayError::Internal(e.to_string()))
}
```
